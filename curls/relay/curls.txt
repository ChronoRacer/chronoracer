# Start

curl -X POST https://relay.chronoracer.live/race/start \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $OKTA_ACCESS_TOKEN" \
  -d "@race_start_request.json"

# Status

curl -X GET https://relay.chronoracer.live/race/status -H "Authorization: Bearer $OKTA_ACCESS_TOKEN"

# Graceful stop

curl -X POST https://relay.chronoracer.live/race/stop \
  -H "Authorization: Bearer $OKTA_ACCESS_TOKEN"

# Force destroy

curl -X POST https://relay.chronoracer.live/race/force-destroy \
    -H "Authorization: Bearer $OKTA_ACCESS_TOKEN"

# History

raceId=$(jq -r '.raceId' race_start_request.json)
curl -X GET "https://relay.chronoracer.live/race/history/$raceId" -H "Authorization: Bearer $OKTA_ACCESS_TOKEN"

# History with export

raceId=$(jq -r '.raceId' race_start_request.json)
curl -X GET "https://relay.chronoracer.live/race/history/${raceId}?export_race=true" -H "Authorization: Bearer $OKTA_ACCESS_TOKEN"

# Operator event (raw bytes, Kafka key)

curl -X POST "https://relay.chronoracer.live/race/event?key=FLAG" \
  -H "Authorization: Bearer $OKTA_ACCESS_TOKEN" \
  --data-binary $'opaque-bytes'
